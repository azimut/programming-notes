https://www.graphviz.org/doc/info/lang.html

#+CAPTION: undirected graph
#+begin_src dot
graph G {
  A [ color=red ]
  A -- B
}
#+end_src

#+CAPTION: directed graph
#+begin_src dot
digraph G {
    node [style=filled, color=red]
    edge [style=dotted, color=yellow=
    Hello -> World;
}
#+end_src
