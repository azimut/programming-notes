** Example 2

#+begin_src pov
  #include"colors.inc"
  #include"textures.inc"
  #default{finish{Phong_Glossy}}
  camera{ location<0,800,-500> direction<0,0,1> up<0,1,0> right<4/3,0,0> look_at<500,0,0> }
  light_source { <400, 400,  100> color White }
  light_source { <400, 400, -200> color White }
  light_source { <400,-400,  100> color White }
  background { Black }
  #declare n=40;
  #declare L=140;
  #declare sticks=union {
    #declare k=1;
    #while(k<=n)
      cylinder { <0,-L,0><0,L,0>,10
        pigment { checker rgb<1,k/n,0>,rgb<k/n,0,1-k/n> scale 10}
        rotate    <10*k,0,0>
        translate <40*k,0,0>
        #declare k=k+1
      }
    #end
  }
  object { sticks rotate <clock*360,0,0> translate <0,0,0> }
#+end_src
