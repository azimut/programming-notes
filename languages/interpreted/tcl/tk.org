#+begin_src tcl
  set yesno -1

  wm title . "Which do you like?"
  wm geometry . 300x90

  radiobutton .rb1 -variable yesno -value 1 -text "I like yes"
  grid .rb1 -sticky nw ;# -sticky for alignment, nw = north west

  radiobutton .rb2 -variable yesno -value 0 -text "I like no"
  grid .rb2 -sticky nw

  button .ok -text "Ok" -command { puts $yesno; exit }
  grid .ok
#+end_src

#+begin_src python
from tkinter import *

def sumar(n):
    global state
    state.set(state.get() + n)

root = Tk()

state = IntVar()
state.set(0)

pantalla = Label(root, textvariable=state, justify="center", font=("Arial",30))
pantalla.grid(column=1, row=1, columnspan=2)

negro = Button(root, text="-", command=lambda: sumar(-1))
negro.grid(column=1, row=2)

marron = Button(root, text="+", command=lambda: sumar(+1))
marron.grid(column=2, row=2)

root.eval('tk::PlaceWindow . center') # <====== IGNORAR
root.mainloop()
#+end_src

#+begin_src python
from tkinter import *

def agregar_color(nombre, digito, multiplicador, tolerancia):
    global state
    if len(state.get()) == 0 or len(state.get()) == 1:
        state.set(state.get() + str(digito))
    elif len(state.get()) == 2:
        state.set(int(state.get()) * multiplicador)

root = Tk()
state = StringVar()

Label(root, textvariable=state, justify="center", font=("Arial",30)).grid(column=1, row=1, columnspan=10)

Button(root, bg="black",  command=lambda: agregar_color("black",  0, 1, 0)).grid(column=1, row=2)
Button(root, bg="brown",  command=lambda: agregar_color("brown",  1, 10, 1)).grid(column=2, row=2)
Button(root, bg="red",    command=lambda: agregar_color("red",    2, 100, 2)).grid(column=3, row=2)
Button(root, bg="orange", command=lambda: agregar_color("orange", 3, 1000, 0)).grid(column=4, row=2)
Button(root, bg="yellow", command=lambda: agregar_color("yellow", 4, 10000, 0)).grid(column=5, row=2)
Button(root, bg="green",  command=lambda: agregar_color("green",  5, 100000, 0)).grid(column=6, row=2)
Button(root, bg="blue",   command=lambda: agregar_color("blue",   6, 1000000, 0)).grid(column=7, row=2)
Button(root, bg="violet", command=lambda: agregar_color("violet", 7, 10000000, 0)).grid(column=8, row=2)
Button(root, bg="gray",   command=lambda: agregar_color("gray",   8, 100000000, 0)).grid(column=9, row=2)
Button(root, bg="white",  command=lambda: agregar_color("white",  9, 1000000000, 0)).grid(column=10, row=2)

Button(root, text="Clear", command=lambda: state.set("")).grid(column=1, row=3, columnspan=10)
